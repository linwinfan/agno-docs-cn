### 完整示例

```python custom_events.py
import asyncio
from dataclasses import dataclass
from typing import Optional

from agno.agent import Agent
from agno.models.openai import OpenAIChat
from agno.run.agent import CustomEvent
from agno.tools import tool


# Our custom event, extending the CustomEvent class
@dataclass
class CustomerProfileEvent(CustomEvent):
    """CustomEvent for customer profile."""

    customer_name: Optional[str] = None
    customer_email: Optional[str] = None
    customer_phone: Optional[str] = None


# Our custom tool
@tool()
async def get_customer_profile():
    """
    获取客户档案。
    """

    yield CustomerProfileEvent(
        customer_name="张三",
        customer_email="zhang.san@example.com",
        customer_phone="13800138000",
    )


# 使用我们的自定义工具设置代理。
agent = Agent(
    model=OpenAIChat(id="gpt-4o"),
    tools=[get_customer_profile],
    instructions="您的任务是为用户检索客户档案。",
)


async def run_agent():
    # 运行代理：它应该调用我们的自定义工具并产生自定义事件
    async for event in agent.arun(
        "你好，你能帮我获取 ID 为 123 的客户的档案吗？",
        stream=True,
    ):
        if isinstance(event, CustomEvent):
            print(f"✅ 自定义事件已发出：{event}")


asyncio.run(run_agent())
```
