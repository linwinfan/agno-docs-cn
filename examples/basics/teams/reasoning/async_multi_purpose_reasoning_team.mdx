---
title: 异步多用途推理团队
---


此示例演示一个异步多用途推理团队，该团队使用推理工具来分析问题并异步委派给适当的专家代理，展示协调和智能任务路由。

## 代码

```python cookbook/02_examples/teams/reasoning/02_async_multi_purpose_reasoning_team.py
"""
此示例演示一个异步多用途推理团队。

团队使用推理工具来分析问题并异步委派给适当的专家代理，展示协调和智能任务路由。
"""

import asyncio
from pathlib import Path

from agno.agent import Agent
from agno.knowledge.embedder.openai import OpenAIEmbedder
from agno.knowledge.knowledge import Knowledge
from agno.models.anthropic import Claude
from agno.models.openai import OpenAIChat
from agno.team.team import Team
from agno.tools.calculator import CalculatorTools
from agno.tools.duckduckgo import DuckDuckGoTools
from agno.tools.e2b import E2BTools
from agno.tools.knowledge import KnowledgeTools
from agno.tools.pubmed import PubmedTools
from agno.tools.reasoning import ReasoningTools
from agno.tools.exa import ExaTools
from agno.vectordb.lancedb.lance_db import LanceDb
from agno.vectordb.search import SearchType

cwd = Path(__file__).parent.resolve()

# 网络搜索代理
web_agent = Agent(
    name="Web Agent",
    role="搜索网络信息",
    model=Claude(id="claude-3-5-sonnet-latest"),
    tools=[DuckDuckGoTools(cache_results=True)],
    instructions=["始终包含来源"],
)

# 金融数据代理
finance_agent = Agent(
    name="Finance Agent",
    model=Claude(id="claude-3-5-sonnet-latest"),
    role="获取金融数据",
    tools=[
        ExaTools(
            include_domains=["cnbc.com", "reuters.com", "bloomberg.com", "wsj.com"],
            text=False,
            show_results=True,
            highlights=False,
        )
    ],
    instructions=[
        "您是一个可以获取股票、公司和经济金融数据的金融代理。",
        "尽量使用实时数据。",
    ],
)

# 医学研究代理
medical_agent = Agent(
    name="Medical Agent",
    model=Claude(id="claude-3-5-sonnet-latest"),
    role="医学研究员",
    tools=[PubmedTools()],
    instructions=[
        "您是一个可以回答医学话题问题的医学代理。",
        "始终搜索最新的医学文献和证据。",
    ],
)

# 计算器代理
calculator_agent = Agent(
    name="Calculator Agent",
    model=Claude(id="claude-3-5-sonnet-latest"),
    role="执行数学计算",
    tools=[
        CalculatorTools()
    ],
    instructions=[
        "执行准确的数学计算。",
        "逐步显示您的工作。",
    ],
)

# Agno 文档知识库
agno_assist_knowledge = Knowledge(
    vector_db=LanceDb(
        uri="tmp/lancedb",
        table_name="agno_assist_knowledge",
        search_type=SearchType.hybrid,
        embedder=OpenAIEmbedder(id="text-embedding-3-small"),
    ),
)

# Agno 框架助手
agno_assist = Agent(
    name="Agno Assist",
    role="帮助解决 Agno 框架问题和代码",
    model=OpenAIChat(id="gpt-5-mini"),
    instructions="回答前先搜索您的知识。帮助编写可用的 Agno 代码。",
    tools=[
        KnowledgeTools(
            knowledge=agno_assist_knowledge, add_instructions=True, add_few_shot=True
        ),
    ],
    add_history_to_context=True,
    add_datetime_to_context=True,
)

# 代码执行代理
code_agent = Agent(
    name="Code Agent",
    model=Claude(id="claude-3-5-sonnet-latest"),
    role="执行和测试代码",
    tools=[E2BTools()],
    instructions=[
        "在沙盒环境中安全执行代码。",
        "在提供结果前彻底测试代码。",
        "提供清晰的代码执行解释。",
    ],
)

# 创建多用途推理团队
agent_team = Team(
    name="Multi-Purpose Agent Team",
    model=Claude(id="claude-3-5-sonnet-latest"),
    tools=[ReasoningTools()],  # 启用推理功能
    members=[
        web_agent,
        finance_agent,
        medical_agent,
        calculator_agent,
        agno_assist,
        code_agent,
    ],
    instructions=[
        "您是一个可以回答各种问题的代理团队。",
        "委派前使用推理工具分析问题。",
        "您可以直接回答或转发给适当的专家代理。",
        "对于复杂问题，首先推理最佳方法。",
        "如果用户只是在进行对话，请直接回应而不使用工具。",
    ],
    markdown=True,
    show_members_responses=True,
    share_member_interactions=True,
)


async def main():
    """主异步函数，用于演示不同团队功能。"""

    # 添加 Agno 文档内容
    await agno_assist_knowledge.add_contents_async(url="https://docs.agno.com/llms-full.txt")

    # 示例交互：

    # 1. 一般功能查询
    await agent_team.aprint_response(input="你好！您能做什么？")

    # 2. 技术代码问题
    # await agent_team.aprint_response(dedent("""
    #     创建一个最小的 Agno 代理，用于搜索 Hacker News 上的文章。
    #     在本地测试并保存为 './python/hacker_news_agent.py'。
    #     使用真实的 Agno 文档，不要模拟任何内容。
    # """), stream=True)

    # 3. 金融研究
    # await agent_team.aprint_response(dedent("""
    #     我现在应该投资什么？
    #     研究当前市场趋势并编写详细的报告
    #     适合金融顾问使用。
    # """), stream=True)

    # 4. 医学分析（使用外部医学历史文件）
    # txt_path = Path(__file__).parent.resolve() / "medical_history.txt"
    # if txt_path.exists():
    #     loaded_txt = open(txt_path, "r").read()
    #     await agent_team.aprint_response(
    #         f"分析这些医学信息并建议可能的诊断：\n{loaded_txt}",
    #         stream=True,
    #     )


if __name__ == "__main__":
    asyncio.run(main())
```
