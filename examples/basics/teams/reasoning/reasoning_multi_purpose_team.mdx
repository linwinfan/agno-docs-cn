title: 多用途推理团队

此示例演示一个综合的专家代理团队，该团队使用推理工具来分析问题并将任务智能地委派给适当的成员，包括网络搜索、金融、写作、医学搜索和代码执行代理。

## 代码

```python cookbook/02_examples/teams/reasoning/01_reasoning_multi_purpose_team.py
"""
此示例演示一个可以回答各种问题的代理团队。

团队使用推理工具来推理问题并委派给适当的代理。

团队组成包括：
- 可以搜索网络信息的网络代理
- 可以获取金融数据的金融代理
- 可以编写内容的写作代理
- 可以进行计算的计算器代理
- 可以解释如何编写 FastAPI 代码的 FastAPI 助手
- 可以在安全 E2B 沙盒中执行代码的代码执行代理
"""

import asyncio
from pathlib import Path
from textwrap import dedent

from agno.agent import Agent
from agno.knowledge.embedder.openai import OpenAIEmbedder
from agno.knowledge.knowledge import Knowledge
from agno.models.anthropic import Claude
from agno.models.openai import OpenAIChat
from agno.team import Team
from agno.tools.calculator import CalculatorTools
from agno.tools.duckduckgo import DuckDuckGoTools
from agno.tools.file import FileTools
from agno.tools.github import GithubTools
from agno.tools.knowledge import KnowledgeTools
from agno.tools.pubmed import PubmedTools
from agno.tools.python import PythonTools
from agno.tools.reasoning import ReasoningTools
from agno.tools.exa import ExaTools
from agno.vectordb.lancedb import LanceDb
from agno.vectordb.search import SearchType

cwd = Path(__file__).parent.resolve()

# 可以搜索网络信息的代理
web_agent = Agent(
    name="Web Agent",
    role="搜索网络信息",
    model=Claude(id="claude-3-5-sonnet-latest"),
    tools=[DuckDuckGoTools(cache_results=True)],
    instructions=["始终包含来源"],
)

reddit_researcher = Agent(
    name="Reddit Researcher",
    role="在 Reddit 上研究一个主题",
    model=Claude(id="claude-3-5-sonnet-latest"),
    tools=[DuckDuckGoTools(cache_results=True)],
    add_name_to_context=True,
    instructions=dedent("""
    您是一个 Reddit 研究员。
    您将获得一个在 Reddit 上研究的主题。
    您需要在 Reddit 上找到最相关的信息。
    """),
)

# 可以获取金融数据的代理
finance_agent = Agent(
    name="Finance Agent",
    role="获取金融数据",
    model=Claude(id="claude-3-5-sonnet-latest"),
    tools=[
        ExaTools(
            include_domains=["cnbc.com", "reuters.com", "bloomberg.com", "wsj.com"],
            text=False,
            show_results=True,
            highlights=False,
        )
    ],
    instructions=["使用表格显示数据"],
)

# 可以编写内容的写手代理
writer_agent = Agent(
    name="Write Agent",
    role="编写内容",
    model=Claude(id="claude-3-5-sonnet-latest"),
    description="您是一个可以编写内容的 AI 代理。",
    instructions=[
        "您是一个多才多艺的写手，可以就任何主题创建内容。",
        "当给定一个主题时，以请求的格式和风格编写引人入胜且信息丰富的内容。",
        "如果您从计算器代理收到数学表达式或计算，将其转换为清晰的文字。",
        "确保您的写作清晰、准确，并针对特定请求量身定制。",
        "在保持事实准确的同时维持自然、引人入胜的语调。",
        "写一些足够好可以发表在《纽约时报》等报纸上的内容。",
    ],
)

# 可以编写内容的写手代理
medical_agent = Agent(
    name="Medical Agent",
    role="编写内容",
    model=Claude(id="claude-3-5-sonnet-latest"),
    description="您是一个可以编写内容的 AI 代理。",
    tools=[PubmedTools()],
    instructions=[
        "您是一个可以回答医学话题问题的医学代理。",
    ],
)

# 可以进行计算的计算器代理
calculator_agent = Agent(
    name="Calculator Agent",
    model=Claude(id="claude-3-5-sonnet-latest"),
    role="计算",
    tools=[
        CalculatorTools()
    ],
)

agno_assist_knowledge = Knowledge(
    vector_db=LanceDb(
        uri="tmp/lancedb",
        table_name="agno_assist_knowledge",
        search_type=SearchType.hybrid,
        embedder=OpenAIEmbedder(id="text-embedding-3-small"),
    ),
)

agno_assist = Agent(
    name="Agno Assist",
    role="您帮助回答关于 Agno 框架的问题。",
    model=OpenAIChat(id="gpt-5-mini"),
    instructions="回答问题前先搜索您的知识。帮助我为 Agno 代理编写可运行的代码。",
    tools=[
        KnowledgeTools(
            knowledge=agno_assist_knowledge, add_instructions=True, add_few_shot=True
        ),
    ],
    add_history_to_context=True,
    add_datetime_to_context=True,
)

github_agent = Agent(
    name="Github Agent",
    role="对 Github 仓库进行分析",
    model=OpenAIChat(id="gpt-5-mini"),
    instructions=[
        "使用您的工具回答关于仓库 agno-agi/agno 的问题",
        "除非明确要求，否则不要创建任何问题或拉取请求",
    ],
    tools=[
        GithubTools(
            list_issues=True,
            list_issue_comments=True,
            get_pull_request=True,
            get_issue=True,
            get_pull_request_comments=True,
        )
    ],
)

local_python_agent = Agent(
    name="Local Python Agent",
    role="在本地运行 Python 代码",
    model=OpenAIChat(id="gpt-5-mini"),
    instructions=[
        "使用您的工具在本地运行 Python 代码",
    ],
    tools=[
        FileTools(base_dir=cwd),
        PythonTools(
            base_dir=Path(cwd), list_files=True, run_files=True, uv_pip_install=True
        ),
    ],
)


agent_team = Team(
    name="多用途团队",
    model=Claude(id="claude-3-7-sonnet-latest"),
    tools=[
        ReasoningTools(add_instructions=True, add_few_shot=True),
    ],
    members=[
        web_agent,
        finance_agent,
        writer_agent,
        calculator_agent,
        agno_assist,
        github_agent,
        local_python_agent,
    ],
    instructions=[
        "您是一个可以回答各种问题的代理团队。",
        "您可以使用成员代理来回答问题。",
        "您也可以直接回答，不必将问题转发给成员代理。",
        "在委派给成员代理之前，先对更复杂的问题进行推理。",
        "如果用户只是在进行对话，请不要使用任何工具，直接回答即可。",
    ],
    markdown=True,
    show_members_responses=True,
    share_member_interactions=True,
)

if __name__ == "__main__":
    # Load the knowledge base
    asyncio.run(
        agno_assist_knowledge.add_contents_async(url="https://docs.agno.com/llms-full.txt")
    )

    # asyncio.run(agent_team.aprint_response("你好！您能做什么？"))

    # Python 代码执行
    # asyncio.run(agent_team.aprint_response(dedent("""实现一个搜索 Hacker News 上好文章的 Agno 代理的正确方法是什么？
    #                                        为我创建一个最小示例，并在本地测试以确保它不会立即崩溃。
    #                                        将创建的代码保存在名为 `./python/hacker_news_agent.py` 的文件中。
    #                                        不要模拟任何内容。使用 Agno 文档中的真实信息。"""), stream=True))

    # # Reddit 研究
    # asyncio.run(agent_team.aprint_response(dedent("""我现在应该投资什么？
    #                                        找一些热门的子版块并进行自己的研究。
    #                                        编写一份详细的报告，可以提供给金融顾问。"""), stream=True))

    # Github 分析
    # asyncio.run(agent_team.aprint_response(dedent("""列出 agno-agi/agno 仓库中的开放拉取请求。
    #                                        找到一个您认为可以解决的问题，并告诉我问题编号、
    #                                        您的建议解决方案和一些代码片段。"""), stream=True))

    # 医学研究
    txt_path = Path(__file__).parent.resolve() / "medical_history.txt"
    loaded_txt = open(txt_path, "r", encoding="utf-8").read()
    agent_team.print_response(
        input=dedent(f"""我有一个患者，其医学信息如下：\n {loaded_txt}
                         最可能的诊断是什么？
                        """),
    )
```
